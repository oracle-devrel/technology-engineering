declare
    l_context apex_exec.t_context;
    l_sql VARCHAR2(4000);
    l_cursor integer;
    l_profile_name VARCHAR2(256);
    l_cnt number;
    l_conversation_time varchar2(500);
BEGIN

    -- ensure that the prompt id is valid. Return fail if it's not. This should not fail.
    BEGIN

       l_cnt:=0;

       SELECT COUNT(1) into l_cnt
       FROM USER_CLOUD_AI_CONVERSATION_PROMPTS u
       WHERE conversation_prompt_id = apex_application.g_x01 and 
       u.conversation_prompt_id NOT IN (select distinct conversation_prompt_id from ADB_CHAT_PROMPTS where conversation_prompt_id is not null);
       
       IF l_cnt > 0 THEN 
        SELECT 'NL2SQL'||'|'||profile_name ,prompt_response
        INTO  l_profile_name,l_sql
        FROM  user_cloud_ai_conversation_prompts
        WHERE conversation_prompt_id = apex_application.g_x01;
       END IF;

       l_cnt:=0;

       select  count(1) into l_cnt from ADB_CHAT_PROMPTS where conversation_prompt_id = apex_application.g_x01;

       IF l_cnt >0 then
        SELECT  'RAGSQL'||'|'||profile_name,prompt_response INTO l_profile_name,l_sql FROM ADB_CHAT_PROMPTS WHERE conversation_prompt_id = apex_application.g_x01;
       END IF;

        EXCEPTION
            WHEN OTHERS THEN
                -- prompt is not valid
                apex_json.open_object;
                apex_json.write('success', false);
                apex_json.write('error', 'invalid prompt id');
                apex_json.close_object;
                -- bail if there is a prompt id error
                RETURN;

    END;

      BEGIN
        SELECT 
          EXTRACT(MINUTE FROM (end_time - start_time)) || ' Minutes ' ||
          TRUNC(EXTRACT(SECOND FROM (end_time - start_time))) || ' Seconds'
        INTO l_conversation_time
        FROM conversation_time
        WHERE conversation_prompt_id = apex_application.g_x01
        ORDER BY start_time DESC
        FETCH FIRST 1 ROW ONLY;
      EXCEPTION
        WHEN NO_DATA_FOUND THEN
          l_conversation_time := '0 Minutes 0 Seconds';
      END;

    -- Check if the SQL is valid. If it's not, produce an error response.
    BEGIN
        -- parse the sql to see if it's valid
        l_cursor := dbms_sql.open_cursor;
        dbms_sql.parse(l_cursor, l_sql, DBMS_SQL.NATIVE);
        dbms_sql.close_cursor(l_cursor);

        -- continue if successful
        
        EXCEPTION
            WHEN OTHERS THEN
                -- The SQL in the prompt_response is not valid. Produce an error message.
                l_sql := 'SELECT ''Sorry, unable to generate a query from your question.'' AS Response FROM DUAL';
                dbms_sql.close_cursor(l_cursor);
                
    END;

    l_context := apex_exec.open_query_context( 
                     p_location   => apex_exec.c_location_local_db,
                     p_sql_query  => l_sql,
                     p_max_rows => 50,
                     p_total_row_count => true);

   apex_json.open_object;
   apex_json.write('id',apex_application.g_x01 );
   apex_json.write('totalRowCount',apex_exec.get_total_row_count(l_context) );
   apex_json.write('profile_name',l_profile_name);
   apex_json.write('conversation_time',l_conversation_time); 
   apex_json.write('sql',l_sql);
   apex_json.write_context( p_name => 'results', p_context => l_context, p_write_null=> true);
   apex_json.close_object;

 END;
